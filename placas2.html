import cv2
import pytesseract

pytesseract.pytesseract.tesseract_cmd = r"C:\Users\MATUTINO\AppData\Local\Programs\Tesseract-OCR\tesseract.exe"

Lista de placas autorizadas (puedes cambiar los valores):
placas_autorizadas = ["12AB34", "KBH659", "XYZ789", "EAG209","93EINE","BRBH279"]

imagen1 = cv2.imread("placa11.jpg")
imagen2 = cv2.imread("placas2.jpg")

placas_leidas = []

for img in [imagen1, imagen2]:
    if img is None:
        print("⚠️ No se pudo cargar una imagen.")
        continue

    img = cv2.resize(img, (800, 300))

    gris = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

    _, binaria = cv2.threshold(gris, 150, 255, cv2.THRESH_BINARY)

    texto = pytesseract.image_to_string(binaria, lang="eng")

    texto_limpio = texto.strip().replace(" ", "").replace("\n", "").replace("-", "").replace(".", "")
    placas_leidas.append(texto_limpio)

    cv2.imshow("Imagen analizada", binaria)
    print("📝 Texto detectado:", texto_limpio)
    cv2.waitKey(1000)
    cv2.destroyAllWindows()

for placa in placas_leidas:
    if placa in placas_autorizadas:
        print(f"✅ {placa}: ACCESO PERMITIDO")
    else:
        print(f"❌ {placa}: ACCESO DENEGADO")

print("\nPlacas leídas:", placas_leidas)

placas_restantes = [p for p in placas_autorizadas if p not in placas_leidas]
print("Placas no escaneadas:", placas_restantes)
